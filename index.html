// Výpočet dynamického přiblížení, aby mapa pokryla celý displej bez bílých pruhů setTimeout(() => { const mapSize = leafletMap.getSize(); const cw = mapSize.x > 0 ? mapSize.x : window.innerWidth; const ch = mapSize.y > 0 ? mapSize.y : window.innerHeight; // Měřítko potřebné k pokrytí šířky nebo výšky const scaleX = cw / w; const scaleY = ch / h; const coverScale = Math.max(scaleX, scaleY); // Přepočet scale na Leaflet zoom (crs: Simple => zoom 0 = scale 1) const coverZoom = Math.log(coverScale) / Math.LN2; // Přidáme trochu extra zoomu podle orientace // Pokud je na výšku (mobil), přiblížíme trochu více než na šířku (PC) let extraZoom = (cw < ch) ? 1.5 : 0.8; let targetZoom = coverZoom + extraZoom; // Kontrola limitů zoomu if (targetZoom > leafletMap.getMaxZoom()) targetZoom = leafletMap.getMaxZoom(); if (targetZoom < leafletMap.getMinZoom()) targetZoom = leafletMap.getMinZoom(); leafletMap.setView(myCenterPoint, targetZoom); }, 50);
